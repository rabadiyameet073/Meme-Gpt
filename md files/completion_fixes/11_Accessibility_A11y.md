## Step 1: AI-Generated Alt Text

Alt text is generated by BLIP during indexing (see Guide 03). Store it on the Meme model.

### 1.1 Add `alt_text` Column to Meme Model

**Add to:** `backend/app/database.py` (in the `Meme` class)

```python

alt_text = Column(Text, nullable=True)  # AI-generated alt text from BLIP
```

### 1.2 Generate Alt Text During Indexing

In the `full_index_pipeline.py` (from Guide 03), the BLIP caption is already generated. Map it to `alt_text`:

```python
# In the pipeline loop:
alt_text = img_result.get("caption", "")
if not alt_text:
    alt_text = f"Meme: {meme.get('name', 'image')} — {', '.join(meme.get('emotions', ['funny']))}"
```

### 1.3 Include Alt Text in API Responses

**Modify:** `backend/app/database.py` → `Meme.to_dict()`

Add `"alt_text"` to the returned dictionary:
```python
def to_dict(self) -> dict:
    return {
        # ... existing fields ...
        "alt_text": self.alt_text or f"Meme: {self.name}",
    }
```

### 1.4 Backfill Alt Text for Existing Memes

**File (create new):** `scripts/backfill_alt_text.py`

```python
"""Backfill alt_text for memes that don't have one."""
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent.parent / "backend"))

from app.database import SessionLocal, Meme

db = SessionLocal()
memes = db.query(Meme).filter(Meme.alt_text == None).all()

for meme in memes:
    parts = [f"Meme titled {meme.name}"]
    if meme.emotions_list():
        parts.append(f"expressing {', '.join(meme.emotions_list())}")
    if meme.categories_list():
        parts.append(f"about {', '.join(meme.categories_list())}")
    if meme.dialogue:
        parts.append(f"with text: {meme.dialogue[:100]}")
    meme.alt_text = ". ".join(parts)

db.commit()
print(f"✅ Backfilled alt_text for {len(memes)} memes")
db.close()
```

## Step 4: Keyboard Navigation

### 4.1 Focus Trap for Modals

**File (create new):** `frontend/src/hooks/useFocusTrap.ts`

```typescript
import { useEffect, useRef } from 'react';

export function useFocusTrap(isActive: boolean) {
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!isActive || !containerRef.current) return;

    const container = containerRef.current;
    const focusable = container.querySelectorAll<HTMLElement>(
      'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
    );
    const first = focusable[0];
    const last = focusable[focusable.length - 1];

    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key !== 'Tab') return;
      if (e.shiftKey) {
        if (document.activeElement === first) {
          e.preventDefault();
          last?.focus();
        }
      } else {
        if (document.activeElement === last) {
          e.preventDefault();
          first?.focus();
        }
      }
    };

    container.addEventListener('keydown', handleKeyDown);
    first?.focus();

    return () => container.removeEventListener('keydown', handleKeyDown);
  }, [isActive]);

  return containerRef;
}
```

## Step 5: Color Contrast Audit

### 5.1 Check Current Contrast Ratios

Run Lighthouse accessibility audit:
```powershell
npx lighthouse http://localhost:5173 --only-categories=accessibility --output html
```

### 5.2 Fix Common Issues

In `frontend/src/index.css`, ensure all text/background combos have ≥ 4.5:1 ratio:

```css
:root {
  /* High contrast text colors */
  --text-primary: #F5F5F5;      /* White text on dark: 15.3:1 ✅ */
  --text-secondary: #A3A3A3;    /* Gray text on dark: 7.5:1 ✅ */
  --text-tertiary: #737373;     /* Light gray: 4.6:1 ✅ (border) */
  /* --text-disabled: #525252;    3.0:1 ❌ — increase to #6B6B6B for 4.5:1 */
  --text-disabled: #6B6B6B;     /* Fixed: 4.5:1 ✅ */

  --bg-primary: #0A0A0A;
  --bg-secondary: #141414;

  /* Brand colors on dark bg */
  --brand-primary: #7C3AED;     /* Purple on dark: 4.7:1 ✅ */
  --brand-accent: #A855F7;      /* Light purple: 5.2:1 ✅ */
}
```
